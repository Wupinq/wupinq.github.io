ZERO = "\u200b"
ONE = "\u200c"

def hide(cover, secret):
    bits = "".join(f"{b:08b}" for b in secret.encode("utf-8"))
    ghost = "".join(ONE if bit == "1" else ZERO for bit in bits)
    return cover[:1] + ghost + cover[1:]

def reveal(text):
    bits = "".join("1" if ch == ONE else "0" for ch in text if ch in (ZERO, ONE))
    data = bytes(int(bits[i:i+8], 2) for i in range(0, len(bits) - 7, 8))
    return data.decode("utf-8", errors="ignore")
